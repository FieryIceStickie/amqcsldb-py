import asyncio
import logging
import os
from contextlib import aclosing

from dotenv import load_dotenv
from log import setup_logging

import amqcsl
from amqcsl.exceptions import QuitError
from amqcsl.workflows import character as cm

_ = load_dotenv()

artists: cm.ArtistDict = {}


async def main(logger: logging.Logger) -> None:
    async with amqcsl.AsyncDBClient(
        username=os.getenv('AMQ_USERNAME'),
        password=os.getenv('AMQ_PASSWORD'),
    ) as client:
        artist_to_meta = await cm.make_artist_to_meta(
            client,
            artists,
            [],
        )
        my_group = client.groups['INSERT GROUP NAME HERE']
        tracks = client.iter_tracks(groups=[my_group], batch_size=100)
        async with aclosing(artist_to_meta.iter_edits(tracks)) as edits:
            async for bundle in edits:
                client.enqueue(bundle)

        if cm.prompt(client.queue):
            await client.commit()


if __name__ == '__main__':
    logger = logging.getLogger('TEMPLATE_SCRIPT_NAME')
    setup_logging()
    try:
        asyncio.run(main(logger))
    except* QuitError:
        logger.info('Quit requested, exiting')
